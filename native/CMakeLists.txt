cmake_minimum_required(VERSION 3.22)
project(orgenda LANGUAGES C)
set(CMAKE_C_STANDARD 11)
add_library(orgenda SHARED orgenda.c ../vendor/tree-sitter/lib/src/lib.c ../vendor/tree-sitter-org/parser.c ../vendor/tree-sitter-org/scanner.c)
target_include_directories(orgenda PRIVATE ../vendor/tree-sitter/lib/include ../vendor/tree-sitter/lib/src ../vendor/tree-sitter-org)
if(ANDROID)
    target_link_options(orgenda PRIVATE "-Wl,-z,max-page-size=16384")
    include(images.cmake)
else()
    find_package(JNI REQUIRED)
    target_include_directories(orgenda PRIVATE ${JNI_INCLUDE_DIRS})
endif()
